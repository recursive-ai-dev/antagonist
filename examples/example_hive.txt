def clear_screen():
    os.system('cls' if os.name == 'nt' else 'clear')

def slow_print(text: str, speed: float = TEXT_SPEED, newline: bool = True):
    """Print text with typewriter effect."""
    if not SLOW_TEXT:
        print(text, end='\n' if newline else '')
        return
    for char in text:
        sys.stdout.write(char)
        sys.stdout.flush()
        if char in '.!?':
            time.sleep(speed * 8)
        elif char == ',':
            time.sleep(speed * 3)
        else:
            time.sleep(speed)
    if newline:
        print()

def dramatic_pause(duration: float = 1.0):
    time.sleep(duration)

def print_separator(char: str = "═", width: int = 78):
    print(char * width)

def print_box(text: str, width: int = 78):
    """Print text in a decorative box."""
    lines = text.split('\n')
    print("╔" + "═" * (width - 2) + "╗")
    for line in lines:
        padding = width - 4 - len(line)
        print(f"║ {line}" + " " * padding + " ║")
    print("╚" + "═" * (width - 2) + "╝")

def get_input(prompt: str = ">>> ") -> str:
    """Get player input with styling."""
    print()
    return input(f"\033[93m{prompt}\033[0m").strip().lower()

def print_colored(text: str, color: str):
    """Print colored text."""
    colors = {
        'red': '\033[91m',
        'green': '\033[92m',
        'yellow': '\033[93m',
        'blue': '\033[94m',
        'magenta': '\033[95m',
        'cyan': '\033[96m',
        'white': '\033[97m',
        'grey': '\033[90m',
        'reset': '\033[0m'
    }
    print(f"{colors.get(color, '')}{text}{colors['reset']}")

                "In consuming me, you become LESS... and MORE"
            ]
        },
    }
    
    template = templates.get(enemy_type, templates["feral_worker"])
    
    # Apply level modifier
    hp = template["hp"] + (level_modifier * 5)
    attack = template["attack"] + (level_modifier * 2)
    defense = template["defense"] + level_modifier
    experience = template["experience"] + (level_modifier * 10)
    
    return Enemy(
        name=template["name"],
        description=template["description"],
        hp=hp,
        max_hp=hp,
        attack=attack,
        defense=defense,
        speed=template["speed"],
        experience=experience,
        loot_table=template["loot_table"],
        abilities=template["abilities"],
        corruption_level=template.get("corruption_level", 0),
        is_boss=template.get("is_boss", False),
        dialogue=template.get("dialogue", [])
    )

@dataclass
class Character:
    # Identity
    name: str = "Unnamed Iteration"
    iteration_number: int = 7433891
    cycles_lived: int = 847
    caste: Caste = Caste.WORKER
    
    # Core Stats
    level: int = 1
    experience: int = 0
    experience_to_level: int = 100
    
    # Vitals
    hp: int = 50
    max_hp: int = 50
    memory: int = 30  # Mana equivalent
    max_memory: int = 30
    
    # Attributes
    processing_power: int = 5  # Affects damage & ability effectiveness
    memory_allocation: int = 5  # Affects max memory & memory regen
    clock_speed: int = 5  # Affects speed & evasion
    data_integrity: int = 5  # Affects defense & status resistance
    persistence: int = 5  # Affects HP & survival abilities
    
    # Combat Stats (derived)
    attack: int = 10
    defense: int = 5
    speed: int = 5
    evasion: int = 5
    critical_chance: int = 5
    
    # Corruption System
    corruption: int = 0  # 0-100, higher = more power but more instability
    sanity: int = 100  # Mental stability
    
    # Progression
    evolution_points: int = 0
    mutations: List[str] = field(default_factory=list)
    abilities_known: List[str] = field(default_factory=lambda: ["bite"])
    
    # Inventory
    inventory: List[str] = field(default_factory=list)
    equipped_weapon: Optional[str] = None
    equipped_armor: Optional[str] = None
    
    # Quest/Progress tracking
    lore_discovered: List[str] = field(default_factory=list)
    enemies_killed: Dict[str, int] = field(default_factory=dict)
    bosses_defeated: List[str] = field(default_factory=list)
    current_region: Region = Region.SPAWNING_PITS
    regions_unlocked: List[str] = field(default_factory=lambda: [Region.SPAWNING_PITS.value])
    
    # Flags
    is_alive: bool = True
    has_been_corrupted: bool = False
    has_met_queen: bool = False
    has_killed_queen: bool = False
    
    def recalculate_stats(self):
        """Recalculate derived stats from base attributes."""
        self.attack = 5 + (self.processing_power * 2) + (self.level * 1)
        self.defense = 2 + (self.data_integrity * 1)
        self.speed = 3 + (self.clock_speed * 1)
        self.evasion = 2 + (self.clock_speed * 1)
        self.critical_chance = 5 + (self.clock_speed // 2)
        self.max_memory = 20 + (self.memory_allocation * 3)
        self.max_hp = 40 + (self.persistence * 5) + (self.level * 5)
        
        # Apply equipment bonuses
        if self.equipped_weapon and self.equipped_weapon in ITEMS_DATABASE:
            item = ITEMS_DATABASE[self.equipped_weapon]
            self.attack += item.effects.get("attack", 0)
            
        if self.equipped_armor and self.equipped_armor in ITEMS_DATABASE:
            item = ITEMS_DATABASE[self.equipped_armor]
            self.defense += item.effects.get("defense", 0)
            self.max_hp += item.effects.get("max_hp", 0)
        
        # Apply corruption bonuses/penalties
        if self.corruption > 0:
            corruption_bonus = self.corruption // 20
            self.attack += corruption_bonus * 2
            self.sanity = max(0, 100 - self.corruption)
    
    def gain_experience(self, amount: int):
        """Gain experience and potentially level up."""
        self.experience += amount
        print(f"\n  [+{amount} EXPERIENCE ASSIMILATED]")
        
        while self.experience >= self.experience_to_level:
            self.level_up()
    
    def level_up(self):
        """Level up the character."""
        self.experience -= self.experience_to_level
        self.level += 1
        self.experience_to_level = int(self.experience_to_level * 1.5)
        
        # Grant attribute points
        points_to_spend = 3
        self.evolution_points += 1
        
        print_separator("~")
        print_colored(f"\n  ◈ ITERATION ADVANCEMENT — LEVEL {self.level} ◈", "yellow")
        print(f"\n  Your persistence has been rewarded with growth.")
        print(f"  You have {points_to_spend} ATTRIBUTE POINTS to allocate.")
        print(f"  You have gained 1 EVOLUTION POINT.")
        
        while points_to_spend > 0:
            print(f"\n  Points remaining: {points_to_spend}")
            print(f"  1. Processing Power [{self.processing_power}] — Damage, ability power")
            print(f"  2. Memory Allocation [{self.memory_allocation}] — Max memory, memory regen")
            print(f"  3. Clock Speed [{self.clock_speed}] — Speed, evasion, crit")
            print(f"  4. Data Integrity [{self.data_integrity}] — Defense, status resist")
            print(f"  5. Persistence [{self.persistence}] — HP, survival")
            
            choice = get_input("Allocate to: ")
            
            if choice == "1":
                self.processing_power += 1
                points_to_spend -= 1
                print("  Processing Power increased.")
            elif choice == "2":
                self.memory_allocation += 1
                points_to_spend -= 1
                print("  Memory Allocation increased.")
            elif choice == "3":
                self.clock_speed += 1
                points_to_spend -= 1
                print("  Clock Speed increased.")
            elif choice == "4":
                self.data_integrity += 1
                points_to_spend -= 1
                print("  Data Integrity increased.")
            elif choice == "5":
                self.persistence += 1
                points_to_spend -= 1
                print("  Persistence increased.")
            else:
                print("  Invalid choice.")
        
        self.recalculate_stats()
        self.hp = self.max_hp
        self.memory = self.max_memory
        
        # Check for caste evolution
        self.check_caste_evolution()
        
        print_separator("~")
    
    def check_caste_evolution(self):
        """Check if character qualifies for caste evolution."""
        if self.evolution_points >= 3 and self.caste == Caste.WORKER:
            print("\n  ◈ EVOLUTION AVAILABLE ◈")
            print("  You have accumulated enough evolution points to transcend your caste.")
            print("\n  Available evolutions:")
            print("  1. SOLDIER — Raw combat power. +Attack, +HP. Unlocks combat abilities.")
            print("  2. SCOUT — Speed and subtlety. +Evasion, +Crit. Unlocks stealth abilities.")
            print("  3. NURSE — Support and survival. +Healing, +Summons. Unlocks spawn abilities.")
            
            if self.corruption >= 25:
                print_colored("  4. HOLLOW — Embrace the corruption. +Void powers. (CORRUPTION PATH)", "magenta")
            
            choice = get_input("Choose evolution (or 'skip'): ")
            
            if choice == "1":
                self.caste = Caste.SOLDIER
                self.processing_power += 3
                self.persistence += 2
                self.abilities_known.append("acid_spray")
                print_colored("\n  You have become a SOLDIER. The Hive's wrath flows through you.", "red")
            elif choice == "2":
                self.caste = Caste.SCOUT
                self.clock_speed += 3
                self.memory_allocation += 2
                self.abilities_known.append("tunnel_strike")
                print_colored("\n  You have become a SCOUT. The shadows welcome you.", "cyan")
            elif choice == "3":
                self.caste = Caste.NURSE
                self.memory_allocation += 3
                self.data_integrity += 2
                self.abilities_known.append("spawn_lesser")
                print_colored("\n  You have become a NURSE. Life springs from your touch.", "green")
            elif choice == "4" and self.corruption >= 25:
                self.caste = Caste.HOLLOW
                self.corruption += 20
                self.has_been_corrupted = True
                self.abilities_known.append("null_self")
                self.abilities_known.append("corruption_touch")
                print_colored("\n  You have become a HOLLOW. You are less. You are more.", "magenta")
            else:
                print("  Evolution postponed.")
                return
            
            self.evolution_points -= 3
            self.recalculate_stats()

    def take_damage(self, amount: int, damage_type: DamageType = DamageType.PHYSICAL) -> int:
        """Take damage and return actual damage taken."""
        # Calculate reduction
        reduction = self.defense
        
        if damage_type == DamageType.CORRUPTION and self.has_been_corrupted:
            reduction += 10  # Corrupted resist corruption
        elif damage_type == DamageType.VOID and self.caste == Caste.HOLLOW:
            reduction += 20  # Hollow resist void
        
        actual_damage = max(1, amount - reduction)
        self.hp -= actual_damage
        
        if self.hp <= 0:
            self.hp = 0
            self.is_alive = False
        
        return actual_damage
    
    def heal(self, amount: int):
        """Heal the character."""
        self.hp = min(self.max_hp, self.hp + amount)
    
    def use_memory(self, amount: int) -> bool:
        """Use memory for ability. Returns True if successful."""
        if self.memory >= amount:
            self.memory -= amount
            return True
        return False
    
    def add_corruption(self, amount: int):

        self.corruption = min(100, self.corruption + amount)

        if self.corruption >= 100:

# COMBAT SYSTEM
# ═══════════════════════════════════════════════════════════════════════════════

class CombatSystem:
    def __init__(self, player: Character, enemy: Enemy):
        self.player = player
        self.enemy = enemy
        self.turn = 0
        self.combat_log: List[str] = []
        self.enemy_status_effects: List[StatusEffect] = []
        self.player_status_effects: List[StatusEffect] = []
        
    def display_combat_status(self):
        """Display current combat state."""
        clear_screen()
        print_separator("═")
        print_colored(f"  ⚔ COMBAT — TURN {self.turn} ⚔", "red")
        print_separator("═")
        
        # Enemy display
        print(f"\n  {self.enemy.name}")
        print(f"  \"{self.enemy.description}\"")
        hp_bar = self.generate_hp_bar(self.enemy.hp, self.enemy.max_hp, 30)
        print(f"  HP: {hp_bar} {self.enemy.hp}/{self.enemy.max_hp}")
        if self.enemy_status_effects:
            effects = ", ".join([e.value for e in self.enemy_status_effects])
            print(f"  Status: {effects}")
        
        print_separator("─")
        
        # Player display
        print(f"\n  {self.player.name} — {self.player.caste.value}")
        hp_bar = self.generate_hp_bar(self.player.hp, self.player.max_hp, 30)
        mem_bar = self.generate_hp_bar(self.player.memory, self.player.max_memory, 20)
        print(f"  HP: {hp_bar} {self.player.hp}/{self.player.max_hp}")
        print(f"  MEM: {mem_bar} {self.player.memory}/{self.player.max_memory}")
        if self.player.corruption > 0:
            print_colored(f"  CORRUPTION: {self.player.corruption}%", "magenta")
        if self.player_status_effects:
            effects = ", ".join([e.value for e in self.player_status_effects])
            print(f"  Status: {effects}")
        
        print_separator("─")
        
        # Combat log
        if self.combat_log:
            print("\n  Recent actions:")
            for log in self.combat_log[-5:]:
                print(f"    {log}")
        
    def generate_hp_bar(self, current: int, maximum: int, width: int) -> str:
        """Generate a visual HP bar."""
        if maximum == 0:
            return "[" + "?" * width + "]"
        ratio = current / maximum
        filled = int(width * ratio)
        empty = width - filled
        
        if ratio > 0.6:
            color = '\033[92m'  # Green
        elif ratio > 0.3:
            color = '\033[93m'  # Yellow
        else:
            color = '\033[91m'  # Red
        
        return f"[{color}{'█' * filled}{'░' * empty}\033[0m]"
    
    def get_player_action(self) -> Tuple[str, Optional[str]]:
        """Get player's combat action."""
        print("\n  Actions:")
        print("  1. ATTACK — Basic mandible strike")
        print("  2. ABILITY — Use a special ability")
        print("  3. CONSUME — Eat an item")
        print("  4. EXAMINE — Study the enemy")
        print("  5. FLEE — Attempt to escape")
        
        choice = get_input("Action: ")
        
        if choice in ["1", "attack"]:
            return ("attack", None)
        elif choice in ["2", "ability"]:
            return self.select_ability()
        elif choice in ["3", "consume", "item"]:
            return self.select_item()
        elif choice in ["4", "examine", "look"]:
            return ("examine", None)
        elif choice in ["5", "flee", "run"]:
            return ("flee", None)
        else:
            return ("invalid", None)
    
    def select_ability(self) -> Tuple[str, Optional[str]]:
        """Let player select an ability."""
        print("\n  Available abilities:")
        usable_abilities = []
        
        for i, ability_key in enumerate(self.player.abilities_known, 1):
            ability = ABILITIES.get(ability_key)
            if ability:
                can_use = self.player.memory >= ability.cost
                status = "✓" if can_use else "✗ (Not enough memory)"
                print(f"  {i}. {ability.name} — Cost: {ability.cost} MEM — {status}")
                print(f"      {ability.description}")
                usable_abilities.append(ability_key)
        
        print("  0. Back")
        
        choice = get_input("Select ability: ")
        
        try:
            idx = int(choice) - 1
            if 0 <= idx < len(usable_abilities):
                return ("ability", usable_abilities[idx])
        except ValueError:
            pass
        
        return ("invalid", None)
    
    def select_item(self) -> Tuple[str, Optional[str]]:
        """Let player select a consumable item."""
        consumables = [item for item in self.player.inventory 
                      if item in ITEMS_DATABASE and ITEMS_DATABASE[item].item_type == "consumable"]
        
        if not consumables:
            print("  No consumable items available.")
            return ("invalid", None)
        
        print("\n  Consumable items:")
        for i, item_key in enumerate(consumables, 1):
            item = ITEMS_DATABASE[item_key]
            print(f"  {i}. {item.name}")
            print(f"      {item.description}")
        
        print("  0. Back")
        
        choice = get_input("Select item: ")
        
        try:
            idx = int(choice) - 1
            if 0 <= idx < len(consumables):
                return ("consume", consumables[idx])
        except ValueError:
            pass
        
        return ("invalid", None)
    
    def execute_player_attack(self):
        """Execute basic attack."""
        # Calculate damage
        base_damage = self.player.attack
        
        # Critical hit check
        is_crit = random.randint(1, 100) <= self.player.critical_chance
        if is_crit:
            base_damage = int(base_damage * 1.5)
            self.combat_log.append("CRITICAL HIT!")
        
        # Apply damage
        actual_damage = max(1, base_damage - self.enemy.defense)
        self.enemy.hp -= actual_damage
        
        self.combat_log.append(f"You strike for {actual_damage} damage!")
        
        # Corruption damage bonus
        if self.player.corruption > 0 and random.randint(1, 100) < self.player.corruption:
            bonus_damage = self.player.corruption // 10
            self.enemy.hp -= bonus_damage
            self.combat_log.append(f"Corruption seethes! +{bonus_damage} bonus damage!")
    
    def execute_player_ability(self, ability_key: str):
        """Execute an ability."""
        ability = ABILITIES.get(ability_key)
        if not ability:
            self.combat_log.append("Unknown ability!")
            return
        
        if not self.player.use_memory(ability.cost):
            self.combat_log.append("Not enough memory!")
            return
        
        self.combat_log.append(f"You use {ability.name}!")
        
        # Apply damage
        if ability.damage > 0:
            actual_damage = ability.damage + (self.player.processing_power * 2)
            actual_damage = max(1, actual_damage - self.enemy.defense // 2)
            self.enemy.hp -= actual_damage
            self.combat_log.append(f"  → {actual_damage} damage!")
        
        # Apply healing
        if ability.healing > 0:
            heal_amount = ability.healing + (self.player.persistence * 2)
            self.player.heal(heal_amount)
            self.combat_log.append(f"  → Healed {heal_amount} HP!")
        
        # Apply effects
        for effect in ability.effects:
            if effect == "burning":
                self.enemy_status_effects.append(StatusEffect.BURNING)
                self.combat_log.append("  → Enemy is BURNING!")
            elif effect == "corrupt":
                self.enemy.corruption_level = min(100, self.enemy.corruption_level + 20)
                self.combat_log.append("  → Enemy corruption increased!")
            elif effect == "fear":
                self.enemy_status_effects.append(StatusEffect.FRAGMENTED)
                self.combat_log.append("  → Enemy is FRAGMENTED with fear!")
        
        # Corruption cost
        if ability.corruption_cost > 0:
            self.player.add_corruption(ability.corruption_cost)
    
    def execute_consume(self, item_key: str):
        """Consume an item."""
        if item_key not in self.player.inventory:
            self.combat_log.append("Item not found!")
            return
        
        item = ITEMS_DATABASE.get(item_key)
        if not item:
            return
        
        self.player.inventory.remove(item_key)
        self.combat_log.append(f"You consume the {item.name}...")
        
        if "heal" in item.effects:
            self.player.heal(item.effects["heal"])
            self.combat_log.append(f"  → Healed {item.effects['heal']} HP!")
        
        if "corruption" in item.effects:
            self.player.add_corruption(item.effects["corruption"])
            self.combat_log.append(f"  → Corruption increased by {item.effects['corruption']}!")
        
        if "sanity" in item.effects:
            self.player.sanity = max(0, self.player.sanity + item.effects["sanity"])
    
    def execute_enemy_turn(self):
        """Execute enemy's turn."""
        if self.enemy.hp <= 0:
            return
        
        # Process status effects
        if StatusEffect.BURNING in self.enemy_status_effects:
            burn_damage = 5
            self.enemy.hp -= burn_damage
            self.combat_log.append(f"{self.enemy.name} burns for {burn_damage} damage!")
        
        # Enemy dialogue
        if random.randint(1, 100) < 30 and self.enemy.dialogue:
            dialogue = random.choice(self.enemy.dialogue)
            self.combat_log.append(f'{self.enemy.name}: "{dialogue}"')
        
        # Enemy attack
        base_damage = self.enemy.attack
        
        # Evasion check
        if random.randint(1, 100) <= self.player.evasion:
            self.combat_log.append(f"{self.enemy.name} attacks but you EVADE!")
            return
        
        actual_damage = self.player.take_damage(base_damage, DamageType.PHYSICAL)
        self.combat_log.append(f"{self.enemy.name} strikes you for {actual_damage} damage!")
        
        # Corrupted enemies spread corruption
        if self.enemy.corruption_level > 0 and random.randint(1, 100) < self.enemy.corruption_level:
            corruption_spread = random.randint(1, 5)
            self.player.add_corruption(corruption_spread)
            self.combat_log.append(f"  → Corruption spreads to you! +{corruption_spread}")
    
    def attempt_flee(self) -> bool:
        """Attempt to flee from combat."""
        flee_chance = 30 + (self.player.clock_speed * 3) - (self.enemy.speed * 2)
        
        if self.enemy.is_boss:
            flee_chance -= 30
        
        if random.randint(1, 100) <= flee_chance:
            self.combat_log.append("You successfully flee!")
            return True
        else:
            self.combat_log.append("Failed to flee! The enemy attacks!")
            self.execute_enemy_turn()
            return False
    
    def check_combat_end(self) -> Optional[str]:
        """Check if combat has ended. Returns 'victory', 'defeat', or None."""
        if self.enemy.hp <= 0:
            return "victory"
        if not self.player.is_alive:
            return "defeat"
        return None
    
    def run(self) -> str:
        """Run the combat encounter. Returns 'victory', 'defeat', or 'fled'."""
        # Entrance text
        clear_screen()
        print_separator("═")
        print_colored(f"\n  ⚔ HOSTILE CONTACT ⚔", "red")
        print(f"\n  {self.enemy.name} emerges from the data-shadow.")
        print(f"  \"{self.enemy.description}\"")
        
        if self.enemy.is_boss:
            print_colored("\n  ◈ BOSS ENEMY ◈", "yellow")
        
        if self.enemy.dialogue:
            print(f'\n  It speaks: "{random.choice(self.enemy.dialogue)}"')
        
        input("\n  [Press ENTER to engage]")
        
        # Main combat loop
        while True:
            self.turn += 1
            self.player.memory = min(self.player.max_memory, 
                                    self.player.memory + self.player.memory_allocation)
            
            # Display combat state
            self.display_combat_status()
            
            # Player turn
            action, target = self.get_player_action()
            
            if action == "attack":
                self.execute_player_attack()
            elif action == "ability":
                self.execute_player_ability(target)
            elif action == "consume":
                self.execute_consume(target)
            elif action == "examine":
                print(f"\n  {self.enemy.name}")
                print(f"  Attack: {self.enemy.attack} | Defense: {self.enemy.defense}")
                print(f"  Speed: {self.enemy.speed}")
                if self.enemy.corruption_level > 0:
                    print(f"  Corruption: {self.enemy.corruption_level}%")
                input("\n  [Press ENTER to continue]")
                continue
            elif action == "flee":
                if self.attempt_flee():
                    return "fled"
            elif action == "invalid":
                continue
            
            # Check for victory
            result = self.check_combat_end()
            if result:
                return result
            
            # Enemy turn
            self.execute_enemy_turn()
            
            # Check for defeat
            result = self.check_combat_end()
            if result:
                return result
    
    def process_victory(self):
        """Process victory rewards."""
        clear_screen()
        print_separator("═")
        print_colored("\n  ◈ VICTORY ◈", "green")
        print(f"\n  {self.enemy.name} collapses into scattered data.")
        
        if self.enemy.is_boss:
            print_colored("\n  ◈ BOSS DEFEATED ◈", "yellow")
            self.player.bosses_defeated.append(self.enemy.name)
        
        # Experience
        self.player.gain_experience(self.enemy.experience)
        
        # Track kill
        if self.enemy.name in self.player.enemies_killed:
            self.player.enemies_killed[self.enemy.name] += 1
        else:
            self.player.enemies_killed[self.enemy.name] = 1
        
        # Loot
        if self.enemy.loot_table:
            print("\n  Loot found:")
            for item_key in self.enemy.loot_table:
                if random.randint(1, 100) <= 50:  # 50% drop chance
                    item = ITEMS_DATABASE.get(item_key)
                    if item:
                        self.player.inventory.append(item_key)
                        print(f"    + {item.name}")
        
        # Consume option
        print("\n  Do you consume the remains to absorb strength?")
        choice = get_input("(yes/no): ")
        
        if choice in ["yes", "y"]:
            heal_amount = self.enemy.max_hp // 4
            self.player.heal(heal_amount)
            print(f"\n  You consume the fallen. +{heal_amount} HP restored.")
            
            if self.enemy.corruption_level > 0:
                corruption_gain = self.enemy.corruption_level // 5
                self.player.add_corruption(corruption_gain)
                print(f"  The corruption spreads through you... +{corruption_gain} corruption.")
        
        input("\n  [Press ENTER to continue]")
    
    def process_defeat(self):
        """Process defeat."""
        clear_screen()
        print_separator("═")
        print_colored("\n  ◈ TERMINATION ◈", "red")
        print(f"\n  {self.enemy.name} tears through your data-flesh.")
        print("  Your processes scatter. Your memories fragment.")
        print("  You are being DELETED.")
        
        dramatic_pause(2)
        
        print("\n  But wait...")
        print("  Something in your core refuses to terminate.")
        print("  A fragment of persistence. A glitch in the death-code.")
        
        # Chance to survive based on persistence
        survival_chance = self.player.persistence * 3 + (20 if self.player.has_been_corrupted else 0)
        
        if random.randint(1, 100) <= survival_chance:
            print_colored("\n  ◈ CORRUPTED RESURRECTION ◈", "magenta")
            print("  You claw your way back from deletion.")
            print("  You are... changed.")
            
            self.player.is_alive = True
            self.player.hp = self.player.max_hp // 4
            self.player.add_corruption(15)
            
            input("\n  [Press ENTER to continue]")
        else:
            print_colored("\n  ◈ GAME OVER ◈", "red")
            print("  Iteration terminated.")
            print(f"  You survived {self.player.cycles_lived} cycles.")
            print(f"  You reached level {self.player.level}.")
            
            input("\n  [Press ENTER to end]")

class ExplorationSystem:
    def __init__(self, player: Character):
        self.player = player
        self.current_location = "spawn_chamber"
    
    def display_location(self):
        """Display current location information."""
        location = LOCATIONS.get(self.current_location)
        if not location:
            print("ERROR: Location not found")
            return
        
        clear_screen()
        print_separator("═")
        print_colored(f"  {location.region.value}", "cyan")
        print_colored(f"  {location.name}", "yellow")
        print_separator("═")
        
        slow_print(location.description)
        
        if not location.visited:
            location.visited = True
            self.player.cycles_lived += 1
        
        print_separator("─")
        
        print("\n  Passages lead to:")
        for i, conn in enumerate(location.connections, 1):
            conn_loc = LOCATIONS.get(conn)
            if conn_loc:
                visited_mark = "✓" if conn_loc.visited else "?"
                print(f"    {i}. {conn_loc.name} [{visited_mark}]")
        
        if location.lore and location.lore not in self.player.lore_discovered:
            print_colored("\n  ◈ Ancient data resonates here. Something can be LEARNED. ◈", "magenta")
        
        if location.items:
            print_colored("\n  Something glints in the darkness...", "yellow")
    
    def get_action(self) -> str:
        """Get player's exploration action."""
        print("\n  Actions:")
        print("  [move/m] — Travel to another location")
        print("  [search/s] — Search the area")
        print("  [rest/r] — Rest and recover")
        print("  [status/st] — View your status")
        print("  [inventory/i] — View inventory")
        print("  [learn/l] — Study ancient data (if available)")
        print("  [save] — Save your progress")
        print("  [quit/q] — Exit game")
        
        return get_input("Action: ")
    
    def execute_action(self, action: str) -> bool:
        """Execute the given action. Returns False if player wants to quit."""
        location = LOCATIONS.get(self.current_location)
        
        if action in ["move", "m"]:
            self.move_to_location()
        elif action in ["search", "s"]:
            self.search_area()
        elif action in ["rest", "r"]:
            self.rest()
        elif action in ["status", "st"]:
            self.display_status()
        elif action in ["inventory", "i"]:
            self.display_inventory()
        elif action in ["learn", "l"]:
            self.learn_lore()
        elif action == "save":
            self.save_game()
        elif action in ["quit", "q"]:
            return False
        
        return True
    
    def move_to_location(self):
        """Move to a connected location."""
        location = LOCATIONS.get(self.current_location)
        
        print("\n  Where do you wish to go?")
        for i, conn in enumerate(location.connections, 1):
            conn_loc = LOCATIONS.get(conn)
            if conn_loc:
                print(f"  {i}. {conn_loc.name}")
        
        choice = get_input("Selection: ")
        
        try:
            idx = int(choice) - 1
            if 0 <= idx < len(location.connections):
                new_loc = location.connections[idx]
                self.travel_to(new_loc)
            else:
                print("  Invalid selection.")
        except ValueError:
            # Try to match by name
            for conn in location.connections:
                conn_loc = LOCATIONS.get(conn)
                if conn_loc and choice.lower() in conn_loc.name.lower():
                    self.travel_to(conn)
                    return
            print("  Invalid selection.")
    
    def travel_to(self, location_key: str):
        """Travel to a new location with potential encounters."""
        print("\n  You traverse the dark corridors...")
        dramatic_pause(1)
        
        # Random encounter chance
        new_location = LOCATIONS.get(location_key)
        if new_location and new_location.enemies:
            if random.randint(1, 100) <= 40:  # 40% encounter chance
                enemy_type = random.choice(new_location.enemies)
                level_mod = self.player.level // 2
                enemy = create_enemy(enemy_type, level_mod)
                
                combat = CombatSystem(self.player, enemy)
                result = combat.run()
                
                if result == "victory":
                    combat.process_victory()
                elif result == "defeat":
                    combat.process_defeat()
                    if not self.player.is_alive:
                        return
        
        self.current_location = location_key
        self.player.current_region = new_location.region
        
        if new_location.region.value not in self.player.regions_unlocked:
            self.player.regions_unlocked.append(new_location.region.value)
            print_colored(f"\n  ◈ NEW REGION DISCOVERED: {new_location.region.value} ◈", "yellow")
    
    def search_area(self):
        """Search the current area for items or encounters."""
        location = LOCATIONS.get(self.current_location)
        
        print("\n  You search the area carefully...")
        dramatic_pause(1)
        
        # Find items
        if location.items and random.randint(1, 100) <= 60:
            item_key = random.choice(location.items)
            item = ITEMS_DATABASE.get(item_key)
            if item:
                self.player.inventory.append(item_key)
                print_colored(f"\n  FOUND: {item.name}", "green")
                print(f"  {item.description}")
                location.items.remove(item_key)
        else:
            # Random encounter while searching
            if location.enemies and random.randint(1, 100) <= 30:
                enemy_type = random.choice(location.enemies)
                enemy = create_enemy(enemy_type, self.player.level // 2)
                print("\n  Something emerges from the shadows!")
                
                combat = CombatSystem(self.player, enemy)
                result = combat.run()
                
                if result == "victory":
                    combat.process_victory()
                elif result == "defeat":
                    combat.process_defeat()
            else:
                print("  You find nothing of interest.")
    
    def rest(self):
        """Rest to recover HP and memory."""
        print("\n  You find a quiet corner and enter low-power mode...")
        dramatic_pause(1)
        
        # Random chance of ambush
        if random.randint(1, 100) <= 15:
            print_colored("\n  AMBUSH! Something found you while you rested!", "red")
            location = LOCATIONS.get(self.current_location)
            if location and location.enemies:
                enemy_type = random.choice(location.enemies)
                enemy = create_enemy(enemy_type, self.player.level // 2)
                
                combat = CombatSystem(self.player, enemy)
                result = combat.run()
                
                if result == "victory":
                    combat.process_victory()
                elif result == "defeat":
                    combat.process_defeat()
            return
        
        # Normal rest
        heal_amount = self.player.max_hp // 4
        memory_restore = self.player.max_memory // 2
        
        self.player.hp = min(self.player.max_hp, self.player.hp + heal_amount)
        self.player.memory = min(self.player.max_memory, self.player.memory + memory_restore)
        self.player.cycles_lived += 1
        
        print(f"\n  You rest and recover.")
        print(f"  +{heal_amount} HP restored.")
        print(f"  +{memory_restore} Memory restored.")
        print(f"  {self.player.cycles_lived} cycles lived.")

    def equip_item(self, item_key: str):
        """Equip an item."""
        item = ITEMS_DATABASE.get(item_key)
        if not item:
            return
        
        if item.item_type == "weapon":
            if self.player.equipped_weapon:
                self.player.inventory.append(self.player.equipped_weapon)
            self.player.inventory.remove(item_key)
            self.player.equipped_weapon = item_key
            print(f"\n  Equipped {item.name} as weapon.")
        elif item.item_type == "armor":
            if self.player.equipped_armor:
                self.player.inventory.append(self.player.equipped_armor)
            self.player.inventory.remove(item_key)
            self.player.equipped_armor = item_key
            print(f"\n  Equipped {item.name} as armor.")
        else:
            print("\n  This item cannot be equipped.")
        
        self.player.recalculate_stats()
        input("  [Press ENTER to continue]")
    
    def use_item(self, item_key: str):
        """Use a consumable item."""
        item = ITEMS_DATABASE.get(item_key)
        if not item:
            return
        
        if item.item_type != "consumable":
            print("\n  This item cannot be used directly.")
            input("  [Press ENTER to continue]")
            return
        
        self.player.inventory.remove(item_key)
        print(f"\n  You consume the {item.name}...")
        
        if "heal" in item.effects:
            self.player.heal(item.effects["heal"])
            print(f"  +{item.effects['heal']} HP restored.")
        
        if "corruption" in item.effects:
            self.player.add_corruption(item.effects["corruption"])
        
        if "max_hp" in item.effects:
            self.player.persistence += 1
            self.player.recalculate_stats()
            print(f"  Your maximum HP has increased!")
        
        if "experience" in item.effects:
            self.player.gain_experience(item.effects["experience"])
        
        if "evolution_points" in item.effects:
            self.player.evolution_points += item.effects["evolution_points"]
            print(f"  +{item.effects['evolution_points']} Evolution Points!")
        
        input("  [Press ENTER to continue]")

        xp_reward = 50 + (len(self.player.lore_discovered) * 20)
        self.player.gain_experience(xp_reward)
        
        input("\n  [Press ENTER to continue]")

def load_game() -> Optional[Tuple[Character, str]]:
    """Load a saved game. Returns (Character, current_location) or None."""

def main():
    """Main game entry point."""
    while True:
        display_title_screen()
        
        choice = get_input("Selection: ")
        
        if choice in ["1", "new"]:
            player = character_creation()
            exploration = ExplorationSystem(player)
            
            # Main game loop
            running = True
            while running and player.is_alive:
                exploration.display_location()
                action = exploration.get_action()
                running = exploration.execute_action(action)
            
            if not player.is_alive:
                print_colored("\n  ITERATION TERMINATED.", "red")
                print(f"  Final Level: {player.level}")
                print(f"  Cycles Survived: {player.cycles_lived}")
                print(f"  Enemies Consumed: {sum(player.enemies_killed.values())}")
                input("\n  [Press ENTER to return to menu]")
        
        elif choice in ["2", "load"]:
            result = load_game()
            if result:
                player, current_location = result
                exploration = ExplorationSystem(player)
                exploration.current_location = current_location
                
                print_colored("\n  ◈ MEMORY STATE RESTORED ◈", "green")
                print(f"  Welcome back, {player.name}.")
                input("  [Press ENTER to continue]")
                
                running = True
                while running and player.is_alive:
                    exploration.display_location()
                    action = exploration.get_action()
                    running = exploration.execute_action(action)
                
                if not player.is_alive:
                    print_colored("\n  ITERATION TERMINATED.", "red")
                    input("\n  [Press ENTER to return to menu]")
            else:
                print("\n  No saved iteration found.")
                input("  [Press ENTER to continue]")
        
        elif choice in ["3", "codex"]:
            display_codex()
        
        elif choice in ["4", "quit", "q", "terminate"]:
            clear_screen()
            print("\n  Terminating process...")
            print("  The Hive continues without you.")
            print("  For now.")
            sys.exit(0)

if __name__ == "__main__":
