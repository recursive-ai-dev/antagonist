class Player:
    def __init__(self, name, caste_key, epithet_key):
        self.name = name; self.caste = caste_key; self.epithet = epithet_key
        self.level = 1; self.xp = 0; self.stat_pts = 0
        self.might=3; self.vigor=3; self.celerity=3; self.synapse=3
        c = CASTES[caste_key]
        for s,v in c["s"].items(): setattr(self,s,getattr(self,s)+v)
        e = EPITHETS[epithet_key]
        for s,v in e["b"].items(): setattr(self,s,getattr(self,s)+v)
        self.max_hp=25+self.vigor*3+e.get("hp",0); self.hp=self.max_hp
        self.max_phero=10+self.synapse*2+e.get("mp",0); self.phero=self.max_phero
        self.base_atk=2+self.might+e.get("atk",0); self.base_def=1+self.vigor//2+e.get("dfn",0)
        self.base_spd=3+self.celerity+e.get("spd",0)
        self.weapon=None; self.armor=None
        self.inventory=[("health_potion",2),("byte_dust",5)]
        self.quests={}; self.completed_quests=[]; self.lore_found=[]
        self.room="hub_nest"; self.keys=0; self.turns=0
        self.searched=set(); self.explored_rooms=set()
        self.phero_abilities=get_pheromones(caste_key)
        self.conditions=[]; self.injuries=[]
        self.swarm_dmg=0; self.swarm_turns=0; self.evade_turns=0
        self.stealthed=False

    @property
    def atk(self): return self.base_atk+(self.weapon and ITEMS[self.weapon].get("atk",0) or 0)
    @property
    def dfn(self): return self.base_def+(self.armor and ITEMS[self.armor].get("def",0) or 0)
    @property
    def spd(self): return self.base_spd

    def equip(self, iid):
        if iid not in [i[0] for i in self.inventory]: return
        it=ITEMS[iid]
        if it["t"]=="armor":
            if self.armor: self.inventory.append((self.armor,1))
            self.armor=iid
        elif it["t"]=="weapon":
            if self.weapon: self.inventory.append((self.weapon,1))
            self.weapon=iid
        self.inventory.remove((iid,1))

    def use_item(self, iid):
        for i,(item,qty) in enumerate(self.inventory):
            if item==iid:
                it=ITEMS[iid]
                if it["t"]=="heal":
                    self.hp=min(self.max_hp,self.hp+it["heal"])
                    return f"+{it['heal']} HP"
                elif it["t"]=="pheromone":
                    self.phero=min(self.max_phero,self.phero+it["restore"])
                    return f"+{it['restore']} Phero"
                elif iid=="memory_key":
                    self.keys+=1
                    return f"KEY {self.keys}/3"
                self.inventory[i]=(item,qty-1)
                if qty<=1: self.inventory.pop(i)
                return None
        return None

    def check_xp(self):
        while self.level<20 and self.xp>=XP_TABLE[self.level]:
            self.xp-=XP_TABLE[self.level]; self.level+=1
            self.stat_pts+=3; self.max_phero+=2; self.phero=min(self.phero+2,self.max_phero)
            return True
        return False

    def get_action(self):
        p=self.player; r=self.rooms[p.room]; u=input(f"\n  {C.BWHT}>{C.RST} ").strip().lower()
        if not u: return

        if u in r.exits:
            nr=r.exits[u]
            if nr in self.rooms:
                p.room=nr; p.turns+=1
                nr2=self.rooms[nr]
                if nr not in p.explored_rooms:
                    p.explored_rooms.add(nr)
                    print(f"\n  {C.BYLW}★ New area: {nr2.name}{C.RST}")
                if nr2.combat and nr2.encounters and random.random()<0.35:
                    self.do_combat(get_encounters(nr2.zone,p.level))
                elif nr2.combat and not nr2.encounters and random.random()<0.25:
                    self.do_combat(get_encounters(nr2.zone,p.level))
            else: print(f"\n  {C.DIM}That way is blocked.{C.RST}")
        elif u=="s": self.do_search()
        elif u=="t": self.do_talk()
        elif u=="i": self.show_inv()
        elif u=="c": self.show_stats()
        elif u=="q": self.show_quests()
        elif u in ("save","save_game"):
            if p.room=="hub_sanctuary" or "save-game" in [i[0] for i in p.inventory]:
                self.do_save()
            else: print(f"\n  {C.RED}Can only save at the Hub Sanctuary.{C.RST}")
        elif u=="rest" and p.room=="hub_sanctuary":
            p.hp=p.max_hp; p.phero=p.max_phero
            print(f"\n  {C.BGRN}You rest. HP and Pheromone restored.{C.RST}")
        elif u=="heal" and p.room=="hub_sanctuary":
            p.hp=min(p.max_hp,p.hp+20)
            print(f"\n  {C.BGRN}Nurse Sil tends your wounds. +20 HP.{C.RST}")
        elif u=="?": self.show_help()
        elif u=="map": self.show_map()
        elif u in ("quit","exit","q"):
            pass  # handled in loop
        else:
            amb=[f"The silence stretches.",f"Your antennae twitch. Nothing.",f"The air hums.",
                 f"Something shifts in the distance.",f"You feel small.",f"Data flows somewhere above."]
            print(f"\n  {C.DIM}{random.choice(amb)}{C.RST}")

    def do_search(self):
        p=self.player; r=self.rooms[p.room]; found=False
        if p.room in p.searched and not r.loot:
            print(f"\n  {C.DIM}Nothing left.{C.RST}"); return
        p.searched.add(p.room)
        if r.loot:
            for iid in list(r.loot):
                it=ITEMS.get(iid)
                if it:
                    print(f"\n  {C.BYLW}Found: {it['clr']}{it['n']}{C.RST}")
                    p.inventory.append((iid,1)); found=True
            r.loot=[]
        if r.lore and r.lore not in p.lore_found:
            p.lore_found.append(r.lore)
            print(f"\n  {C.BMAG}═══ LORE ═══{C.RST}")
            print(f"  {C.ITA}{r.lore}{C.RST}"); found=True
        if not found:
            drops=["byte_dust","byte_dust","byte_dust",None,None,None,None]
            d=random.choice(drops)
            if d:
                print(f"\n  {C.DIM}Found: {ITEMS[d]['n']}{C.RST}")
                p.inventory.append((d,1))
            else: print(f"\n  {C.DIM}Nothing.{C.RST}")

    def do_combat(self, enemy):
        p=self.player; e=enemy; first=True
        self.cls()
        print(f"\n  {C.BRED}═══════════════════════════════════════════{C.RST}")
        print(f"  {e['clr']}{C.BLD}⚔ {e['name']} appears!{C.RST}")
        print(f"  {C.DIM}{e['desc']}{C.RST}")
        print(f"  {C.BRED}═══════════════════════════════════════════{C.RST}")
        time.sleep(1)

        while e['hp']>0 and p.hp>0:
            self.cls()
            print(f"\n  {e['clr']}■ {e['name']} {C.DIM}lv{e['lv']}{C.RST}")
            print(f"  HP {self.bar(e['hp'],e['max_hp'])} {e['hp']}/{e['max_hp']}")
            if e['poison']>0: print(f"  {C.BGRN}☠ Poisoned ({e['poison']}/turn){C.RST}")
            if e['charging']: print(f"  {C.BRED}⚠ CHARGING: {e['sp_name']}!{C.RST}")

            print(f"\n  {C.BGRN}■ {p.name}{C.RST} {C.BYLW}{CASTES[p.caste]['n']}{C.RST}")
            print(f"  HP {self.bar(p.hp,p.max_hp)} {p.hp}/{p.max_hp}")
            print(f"  PH {self.bar(p.phero,p.max_phero,20,C.BBLU)} {p.phero}/{p.max_phero}")

            ca=[i for i in p.phero_abilities if p.phero>=i["c"]]
            if ca:
                ab=", ".join(f"{C.BBLU}p{i+1}{C.RST} {a['n']}" for i,a in enumerate(ca))
                print(f"\n  {C.DIM}{ab}{C.RST}")

            u=input(f"\n  {C.BWHT}[a]ttack  [p]hero  [d]efend  [f]lee  [u]se: {C.RST}").strip().lower()

            if u=="a":
                base=p.atk; is_crit=False
                if p.stealthed: base*=2; is_crit=True; p.stealthed=False
                dmg=max(1,base-e['def']+random.randint(-2,2))
                if is_crit: dmg=int(dmg*1.5)
                e['hp']-=dmg
                cm=f" {C.BRED}CRITICAL!{C.RST}" if is_crit else ""
                print(f"\n  {C.BYLW}You strike for {dmg} damage!{C.RST}{cm}")
                if e['hp']<=0: self.combat_win(e); return

            elif u.startswith("p"):
                if not ca: print(f"\n  {C.RED}No pheromones!{C.RST}")
                elif len(u)==1:
                    for i,a in enumerate(ca): print(f"  {C.BWHT}[{i+1}]{C.RST} {C.BBLU}{a['n']}{C.RST} {C.DIM}({a['c']}P){C.RST} — {a['desc']}")
                    try:
                        ch=int(input(f"  {C.DIM}>{C.RST} ").strip())-1
                        if 0<=ch<len(ca): self.use_phero(ca[ch],e)
                    except: pass
                else:
                    try:
                        idx=int(u[1])-1
                        if 0<=idx<len(ca): self.use_phero(ca[idx],e)
                    except: pass
                if e['hp']<=0: self.combat_win(e); return

            elif u=="d":
                p.phero=min(p.max_phero,p.phero+5)
                print(f"\n  {C.BCYN}You brace! +5 Pheromone. Damage halved.{C.RST}")

            elif u=="f":
                if random.random()<0.4:
                    print(f"\n  {C.BYLW}You flee!{C.RST}"); self.wait(); return
                else: print(f"\n  {C.RED}Can't escape!{C.RST}")

            elif u.startswith("u"):
                hi=[(i,it) for i,(iid,_) in enumerate(p.inventory) if ITEMS.get(iid,{}).get("t") in ("heal","pheromone")]
                if not hi: print(f"\n  {C.DIM}No usable items.{C.RST}")
                else:
                    for j,(idx,iid) in enumerate(hi):
                        it=ITEMS[iid]; print(f"  {C.BWHT}[{j+1}]{C.RST} {it['clr']}{it['n']}{C.RST}")
                    try:
                        ch=int(input(f"  {C.DIM}>{C.RST} ").strip())-1
                        if 0<=ch<len(hi):
                            iid=p.inventory[hi[ch][0]][0]; res=p.use_item(iid)
                            if res: print(f"  {C.GRN}{res}{C.RST}")
                    except: pass

            else: print(f"\n  {C.DIM}You hesitate.{C.RST}")

            if e['hp']<=0: self.combat_win(e); return

            # Enemy turn
            print()
            if e['poison']>0:
                e['hp']-=e['poison']; print(f"  {C.BGRN}Poison: {e['poison']} damage!{C.RST}")
                if e['hp']<=0: self.combat_win(e); return

            if e['sp_cd']>0: e['sp_cd']-=1
            if e['charging']:
                dmg=e['sp_dmg']; e['charging']=False
                if "defending" in dir(self) and self._def:
                    dmg//=2; self._def=False
                dmg=max(0,dmg-p.dfn); p.hp-=dmg
                print(f"  {C.BRED}✦ {e['sp_name']}! {dmg} damage!{C.RST}")
                if random.random()<0.15 and dmg>5:
                    inj=random.choice(["Lost Eye (-1 Celerity)","Broken Mandible (-1 Might)",
                        "Cracked Chitin (-1 Vigor)","Fractured Antenna (-1 Synapse)"])
                    if inj not in p.injuries:
                        p.injuries.append(inj)
                        stat={"Lost Eye (-1 Celerity)":"celerity","Broken Mandible (-1 Might)":"might",
                            "Cracked Chitin (-1 Vigor)":"vigor","Fractured Antenna (-1 Synapse)":"synapse"}[inj]
                        setattr(p,stat,max(1,getattr(p,stat)-1))
                        print(f"  {C.BRED}INJURY: {inj}!{C.RST}")
            elif e['sp_cd']==0 and random.random()<0.3:
                e['charging']=True; e['sp_cd']=3
                print(f"  {e['clr']}⚠ {e['name']} charges {e['sp_name']}!{C.RST}")
            else:
                dmg=max(1,e['atk']-p.dfn+random.randint(-1,1))
                if hasattr(self,'_def') and self._def:
                    dmg//=2; self._def=False
                p.hp-=dmg
                print(f"  {e['clr']}{e['name']} strikes for {dmg}!{C.RST}")

            if p.hp<=0: self.do_death(e); return
            self.swarm_turns-=1
            if self.swarm_turns>0 and p.swarm_dmg>0:
                e['hp']-=p.swarm_dmg; print(f"  {C.BYLW}Swarm: {p.swarm_dmg} damage!{C.RST}")
                if e['hp']<=0: self.combat_win(e); return
            p.evade_turns=max(0,p.evade_turns-1)
            if p.stealthed and u!="a": print(f"  {C.DIM}In shadows. Next attack = crit.{C.RST}")
            time.sleep(0.8)

    def use_phero(self, ab, e):
        p=self.player; p.phero-=ab["c"]; fx=ab["fx"]
        print(f"\n  {C.BBLU}✦ {ab['n']}!{C.RST}")

        if fx=="dmg_aoe":
            sc=getattr(p,ab["sc"],3); dmg=ab["bd"]+sc+random.randint(0,3)
            e['hp']-=dmg; print(f"  {C.BRED}{dmg} damage!{C.RST}")
        elif fx=="party_def": print(f"  {C.BCYN}DEF +{ab['v']}!{C.RST}")
        elif fx=="block": print(f"  {C.BCYN}Wall blocks next attack!{C.RST}"); self._def=True
        elif fx=="self_buff": print(f"  {C.BYLW}ATK +{ab['sa']}, DEF {ab['sd']}!{C.RST}")
        elif fx=="enemy_debuff": print(f"  {C.BYLW}Enemy ATK -{ab['v']}!{C.RST}")
        elif fx=="execute":
            if e['hp']<=e['max_hp']*ab['th']:
                sc=getattr(p,ab["sc"],3); dmg=(ab["bd"]+sc)*2
                e['hp']-=dmg; print(f"  {C.BRED}EXECUTED! {dmg}!{C.RST}")
            else: print(f"  {C.DIM}Too healthy to execute.{C.RST}")
        elif fx=="speed": p.phero=min(p.max_phero,p.phero+ab["c"]); print(f"  {C.BCYN}Blinding speed!{C.RST}")

        elif fx=="acid":
            sc=getattr(p,ab["sc"],3); dmg=ab["bd"]+sc; e['hp']-=dmg; e['def']=max(0,e['def']-ab["v"])
            print(f"  {C.BGRN}{dmg} + armor corroded!{C.RST}")
        elif fx=="heal":
            sc=getattr(p,ab["sc"],3); h=ab["bd"]+sc; p.hp=min(p.max_hp,p.hp+h)
            print(f"  {C.BGRN}+{h} HP!{C.RST}")
        elif fx=="analyze":
            print(f"  HP: {e['hp']}/{e['max_hp']}  ATK: {e['atk']}  DEF: {e['def']}  SPD: {e['spd']}")
            print(f"  {C.DIM}{e['desc']}{C.RST}")
        elif fx=="debuff_all":
            sc=getattr(p,ab["sc"],3); dmg=ab["bd"]+sc; e['hp']-=dmg
            e['atk']=max(0,e['atk']-ab["v"]); e['def']=max(0,e['def']-ab["v"])
            print(f"  {C.BRED}{dmg} + all stats -{ab['v']}!{C.RST}")
        elif fx=="stealth": p.stealthed=True; print(f"  {C.DIM}Into the shadows...{C.RST}")
        elif fx=="poison": e['poison']=ab["v"]; print(f"  {C.BGRN}Venom! {ab['v']}/turn {ab['dur']} turns!{C.RST}")
        elif fx=="backstab":
            sc=getattr(p,"celerity",3)+getattr(p,"synapse",3); dmg=ab["bd"]+sc; e['hp']-=dmg
            e['poison']=max(e['poison'],3); print(f"  {C.BRED}{dmg} + bleed!{C.RST}")

            sc=getattr(p,ab["sc"],3); p.swarm_dmg=ab["bd"]+sc; p.swarm_turns=ab["dur"]
            print(f"  {C.BYLW}Swarm! {p.swarm_dmg}/turn!{C.RST}")
        time.sleep(0.8)

    def combat_win(self, e):

    def lvlup(self):

            elif ch=="s": p.synapse+=1; p.max_phero+=2; p.phero+=2; p.stat_pts-=1; print(f"  {C.BBLU}Synapse → {p.synapse}{C.RST}")

    def run(self):
        self.cls()
        while True:
            self.title()
            ch=input(f"\n  {C.DIM}>{C.RST} ").strip()
            if ch=="1":
                print(f"\n  {C.DIM}Load save?{C.RST}"); lch=input(f"  {C.DIM}(y/n): {C.RST}").strip().lower()
                if lch=="y" and self.do_load(): pass
                else: self.create_char()
                break
            elif ch=="2": self.lore_menu()
            elif ch=="3": self.cls(); print(f"\n  {C.DIM}The Colony endures.{C.RST}\n"); return

        self._def=False
        while not self.game_over:
            try:
                self.show_explore()
                self.get_action()
                self.check_win()
            except KeyboardInterrupt:
                print(f"\n\n  {C.DIM}Ctrl+C detected. The Giant stirs...{C.RST}")
                if input(f"  {C.DIM}Quit? (y/n): {C.RST}").strip().lower()=="y": break
                self.player.room="hub_sanctuary"; self.player.hp=self.player.max_hp
            except Exception as ex:
                print(f"\n  {C.RED}Reality fractures: {ex}{C.RST}")
                self.wait()

        if self.won:
            p=self.player
            print(f"\n  {C.BYLW}═══ FINAL STATS ═══{C.RST}")
            print(f"  Name: {p.name}  Level: {p.level}")
            print(f"  Caste: {CASTES[p.caste]['n']}  Epithet: {EPITHETS[p.epithet]['n']}")
            print(f"  Turns: {p.turns}  Keys: {p.keys}/3")
            print(f"  Lore: {len(p.lore_found)}  Rooms: {len(p.explored_rooms)}")
            self.wait()
        self.cls()
        print(f"\n  {C.DIM}The world is warm. The world is electric.\n  The Colony endures.{C.RST}\n")
